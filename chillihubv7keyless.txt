--// Check Local Saved Key (Hidden Persistence)
local function checkSavedKey()
    if pcall(function() return readfile and readfile(KEY_SAVED_NAME) end) then
        local saved = readfile(KEY_SAVED_NAME)
        if saved == CORRECT_KEY then
            return true
        end
    end
    return false
end

if checkSavedKey() then
    pcall(function()
        loadstring(game:HttpGet("https://raw.githubusercontent.com/kairobloc/stealanegg/refs/heads/main/chillihubv7.lua"))()
    end)
    return
end

-- Smooth Sine Wave Shake Animation
local isShaking = false
local function smoothShake()
    if isShaking then return end
    isShaking = true
    
    local originalPos = UDim2.new(0.5, 0, 0.5, 0)
    local offsets = {10, -8, 6, -4, 2, 0}
    
    for _, offset in ipairs(offsets) do
        local tweenInfo = TweenInfo.new(0.04, Enum.EasingStyle.Sine, Enum.EasingDirection.Out)
        local tween = TweenService:Create(MainFrame, tweenInfo, {Position = originalPos + UDim2.new(0, offset, 0, 0)})
        tween:Play()
        tween.Completed:Wait()
    end
    
    isShaking = false
end

        StatusLabel.Text = "Key Verified! Loading script..."
        
        -- Save key silently (Hidden persistence)
        pcall(function()
            if writefile then
                writefile(KEY_SAVED_NAME, CORRECT_KEY)
            end
        end)
        
        task.wait(0.6)
        
        -- Smooth Fade Out
        local fadeTween = TweenService:Create(MainFrame, TweenInfo.new(0.3), {BackgroundTransparency = 1})
        fadeTween:Play()
        fadeTween.Completed:Wait()
        
        ScreenGui:Destroy()
        
        -- Load main script
        pcall(function()
            loadstring(game:HttpGet("https://raw.githubusercontent.com/kairobloc/stealanegg/refs/heads/main/chillihubv7.lua"))()
        end)
    else
        StatusLabel.TextColor3 = Color3.fromRGB(255, 85, 85)
        BannerStroke.Color = Color3.fromRGB(255, 85, 85)
        StatusLabel.Text = "Invalid Key! Please check and try again."
        
        TextBox.Text = "" -- Auto clear input
        smoothShake()
    end
end)
